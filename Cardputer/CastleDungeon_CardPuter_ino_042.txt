// ================================= S E T U P ==================================
void setup() 
{
  auto cfg = M5.config();
  M5Cardputer.begin(cfg);
  M5Cardputer.Speaker.begin();
  delay(500);
  for(int n=0; n<10; n++) {
    M5Cardputer.Display.printf(" %d ", n); 
    M5Cardputer.Speaker.tone(100+100*n, 500); 
    delay(500);
  }
  delay(1000);

  // Create sprite matching your array dimensions (width, height)
  setupSprite(beast, beastData);
  setupSprite(bomb,  bombData);
  setupSprite(door,  doorData);
  setupSprite(fl00r, fl00rData);
  setupSprite(key,   keyData);
  setupSprite(man,   manData);
  setupSprite(pit,   pitData);
  setupSprite(sword, swordData);
  setupSprite(wall,  wallData);

  // Setup maze
  placeItemRoom('*', 3);  // Bombs
  placeItemRoom('&',12);  // Beasts
  placeItem('@', 3);      // Pits
  placeItem('+', 1);      // Key
  placeItem('!', 1);      // Sword

  // Set initial place in maze
  M5Cardputer.Display.setTextSize(2);
  do {
    row = esp_random()%20;
    col = esp_random()%20;
    //M5Cardputer.Display.setCursor(0, 0); 
    //M5Cardputer.Display.printf("%d %d", row, col);
  } while( maze[row][col] != ' ');

  // Testing placement of player
  char ch = maze[row][col];
  updateDisplay( ch );
  delay(1000);
}

void setupSprite(M5Canvas &s, const uint8_t *data) 
{
  s.createSprite(16, 16);
  s.setColorDepth(8);
  s.pushImage(0, 0, 16, 16, data);
  s.setPivot(0, 0);
}

// ----------------------------------------------------------------------------
void placeItem( char c, int n )
{
  int row, col, i;

  for( i=0; i<n; i++ ) {
    do {
      row = esp_random()%20;
      col = esp_random()%20;
    } while( maze[row][col] != ' ');
    maze[row][col] = c;
  }
}

// ----------------------------------------------------------------------------
void placeItemRoom( char c, int n )
{
  int row, col, i;

  for( i=0; i<n; i++ ) {
    do {
      row = esp_random()%20;
      col = esp_random()%20;
    } while( maze[row][col] != '.');
    maze[row][col] = c;
  }
}

// ================================= M A I N ====================================
void loop() 
{
  int  r = 0, c = 0;
  char k, ch;
  
  k = readKeyboard();
  switch( k ) {
    case 59: r = -1;
              break;
    case 46: r = 1;
              break;
    case 44: c = -1;
              break;
    case 47: c = 1;
              break;
  }
  
  ch = maze[row+r][col+c];
  M5Cardputer.Display.setCursor(0, 0);
  M5Cardputer.Display.printf("%c %d %d ", ch, row+r, col+c); 

  switch( ch ) {
    case ' ': 
              if( c!=0 || r!=0 )
                sfxStep();
              break;
    case '#': 
              r = 0;
              c = 0;
              sfxBumped();
              break;
    case '!': 
              sfxFoundit();
              removeItem( row+r, col+c );
              usword = true;
              break;
    case '+': 
              sfxFoundit();
              removeItem( row+r, col+c );
              ukey = true;
              break;
    case '*': 
              sfxFoundit();
              removeItem( row+r, col+c );
              ubombs++;
              if( ubombs==3 ) { 
                castleSaved();
                gameover = true;
              }
              break;
    case '@':
              sfxPitfall();
              if( spell==0 ) {
                pitFall( row+r, col+c );
                gameover = true;
              }
              break;
    case '%':
              if( usword ) 
                killBeast( row+r, col+c );
              else {
                beastLost( row+r, col+c );
                gameover = true;
              }
              sfxBeast();
              break;
    case '-':
              if( ukey ) 
                removeItem( row+r, col+c );
              else {
                doorBumped( row+r, col+c );
                sfxBumped();
                r = 0;
                c = 0;
              }
  }

  row = row + r;
  col = col + c;
  updateDisplay( ch );
  delay(100);
}

// ----------------------------------------------------------------------------
char readKeyboard( void )
{
  M5Cardputer.update(); 
  if (M5Cardputer.Keyboard.isChange()) {
    if (M5Cardputer.Keyboard.isPressed()) {
      Keyboard_Class::KeysState status = M5Cardputer.Keyboard.keysState();
        if (status.word.size() > 0) {
          char k = status.word[0]; // Grab the character
          return k;
        }        
    }
  }
  return '?';
}

void showSprite( char ch, int x, int y ) 
{
  switch( ch ) {
    case '#': wall.pushRotateZoom(x, y, 0, 2.8, 2.8);
              break;
    case ' ':
    case '.': fl00r.pushRotateZoom(x, y, 0, 2.8, 2.8);
              break;
    case '&': beast.pushRotateZoom(x, y, 0, 2.8, 2.8);
              break;
    case '*': bomb.pushRotateZoom(x, y, 0, 2.8, 2.8);
              break;
    case '@': pit.pushRotateZoom(x, y, 0, 2.8, 2.8);
              break;
    case '-': door.pushRotateZoom(x, y, 0, 2.8, 2.8);
              break;
    case '+': key.pushRotateZoom(x, y, 0, 2.8, 2.8);
              break;
    case '!': sword.pushRotateZoom(x, y, 0, 2.8, 2.8);
              break;
    case '$': man.pushRotateZoom(x, y, 0, 2.8, 2.8);
              break;
  }
}

void updateDisplay( char ch ) 
{
  //M5Cardputer.Display.fillScreen(TFT_BLACK);
  // debugging...
  M5Cardputer.Display.setCursor(0, 16);
  M5Cardputer.Display.printf("%c %d %d ", ch, row, col); 
  
  // row-1  
  ch = maze[row-1][col-1];
  showSprite( ch, xoff, 0 );
  ch = maze[row-1][col];
  showSprite( ch, x+xoff, 0 );
  ch = maze[row-1][col+1];
  showSprite( ch, x*2+xoff, 0 );
  // row  
  ch = maze[row][col-1];
  showSprite( ch, xoff, y );
  ch = '$';
  showSprite( ch, x+xoff, y );
  ch = maze[row][col+1];
  showSprite( ch, x*2+xoff, y );
  // row+1  
  ch = maze[row+1][col-1];
  showSprite( ch, xoff, y*2 );
  ch = maze[row+1][col];
  showSprite( ch, x+xoff, y*2 );
  ch = maze[row+1][col+1];
  showSprite( ch, x*2+xoff, y*2 ); 
}
